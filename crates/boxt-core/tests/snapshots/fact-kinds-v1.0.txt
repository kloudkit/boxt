types
  Assignment = {name: string, append: bool, value: string}
  Directive = if | ifdef | ifndef | elif | elifdef | elifndef | else
  FileKind = source | binary | package | service | config
  Frame = {kind: FrameKind, name: string}
  FrameKind = rpm | deb | archive
  GuardChain = {frames: [GuardFrame], has_alternative: bool, region: u32}
  GuardFrame = {directive: Directive, text: string, identifiers: [string], holds_code: bool}
  InputRef = {index: u32, given: string}
  Location = {input: InputRef, frames: [Frame], path: string | null, line: u32 | null, scope: Scope}
  PathText = string
  RedirectOp = write | append | clobber | read | read_write | dup_out | dup_in | write_both | append_both | here_doc | here_string
  Redirection = {fd: u32 | null, op: RedirectOp, target: string}
  Scope = file | function {qualified_name: string, signature: string} | symbol(string) | unit_section {section: string, key: string} | config_key(string) | script(ScriptName)
  ScriptName = pre_transaction | pre_install | post_install | pre_remove | post_remove | post_transaction | pre_remove_transaction | post_remove_transaction | verify | trigger_pre_install | trigger_install | trigger_remove | trigger_post_remove | file_trigger_install | file_trigger_remove | file_trigger_post_remove | transaction_file_trigger_install | transaction_file_trigger_remove | transaction_file_trigger_post_remove | config
